cmake_minimum_required(VERSION 3.20)
project(Synesthesia LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(SYN_SANITIZE "ASan+UBSan" OFF)
option(SYN_TSAN "ThreadSanitizer (nao combina com SYN_SANITIZE)" OFF)
option(SYN_NATIVE "-march=native (desligue para builds reprodutiveis entre maquinas)" OFF)

add_library(syn_flags INTERFACE)
target_include_directories(syn_flags INTERFACE include)
if(MSVC)
  # MSVC nao funde FMA sem /arch:AVX2; nao usar /fp:fast
  target_compile_options(syn_flags INTERFACE /W4 /fp:precise)
else()
  target_compile_options(syn_flags INTERFACE -Wall -Wextra -Wpedantic -Wshadow -Wconversion
    # Determinismo: nada de fast-math, e sem FMA implicito (resultados iguais entre compiladores)
    -fno-fast-math -ffp-contract=off)
  target_compile_options(syn_flags INTERFACE $<$<CONFIG:Release>:-O2>)
  if(SYN_NATIVE)
    target_compile_options(syn_flags INTERFACE -march=native)
  endif()
  if(SYN_TSAN)
    target_compile_options(syn_flags INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(syn_flags INTERFACE -fsanitize=thread)
  endif()
  if(SYN_SANITIZE)
    target_compile_options(syn_flags INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(syn_flags INTERFACE -fsanitize=address,undefined)
  endif()
endif()

enable_testing()
find_package(Threads REQUIRED)
foreach(t test_config test_rng test_determinism test_spsc)
  add_executable(${t} tests/${t}.cpp)
  target_link_libraries(${t} PRIVATE syn_flags Threads::Threads)
  add_test(NAME ${t} COMMAND ${t})
endforeach()

add_executable(bench_step tools/bench_step.cpp)
target_link_libraries(bench_step PRIVATE syn_flags)

# ---- Fase 1: plataforma Windows (janela Win32 + audio WASAPI) ----
if(WIN32)
  add_library(syn_platform STATIC src/platform/win32_window.cpp src/platform/wasapi_out.cpp)
  target_link_libraries(syn_platform PUBLIC syn_flags Threads::Threads user32 gdi32 ole32 avrt winmm uuid)
  add_executable(sandbox tools/sandbox.cpp)
  target_link_libraries(sandbox PRIVATE syn_platform)
endif()
